def package(name='SKILL.md'):
    b=io.BytesIO()
    with zipfile.ZipFile(b,'w') as z:z.writestr(name,'# Example\n실행 지침')
    b.seek(0);return b

def publish(c,h,tid=None,revision=None,**extra):
    m=dict(name='보고서 Skill',desc='보고서를 정리합니다',owner='등록자',org='공통',platform='Claude',type='Skill',category='공통 업무',visibility='전체 공개',guide='ZIP을 풀고 SKILL.md를 확인하세요.',workTags=['주간 보고','실적 정리'],approval=False,audience=[],mode='file',expectedRevision=revision)
    m.update(extra)
    data={'metadata':json.dumps(m),'artifact':(package(), 'skill.zip')}
    return c.post(f'/api/tools/{tid}/versions' if tid else '/api/tools',data=data,headers=h)

def item(c):return c.get('/api/bootstrap').json['items'][0]

def test_publish_persistence_download_dedup_and_modal_like(app):
    c,h=ready(app);r=publish(c,h);assert r.status_code==201,r.json
    assert item(c)['active']==1 and item(c)['revision']==1
    for _ in range(2):assert c.put('/api/tools/1/like',json={'liked':True},headers=h).json['count']==1
    assert c.head('/download/1/1').status_code==200
    assert item(c)['downloads']==0
    for _ in range(2):
        r=c.get('/download/1/1');assert r.status_code==200;assert zipfile.is_zipfile(io.BytesIO(r.data));r.close()
    assert item(c)['downloads']==1
    assert c.put('/api/tools/1/like',json={'liked':False},headers=h).json['count']==0
    second=create_app({**app.config,'TESTING':True});c2,_=ready(second)
    assert item(c2)['workTags']==['주간 보고','실적 정리']
    assert item(c2)['downloads']==1

def test_revision_rollback_protected_delete_and_monotonic_release(app):
    c,h=ready(app);publish(c,h)
    assert publish(c,h,1,1,guide='v2 설치 방법').status_code==201

    assert c.delete('/api/tools/1/versions/2',headers=h).status_code==409
    assert c.post('/api/tools/1/rollback',json={'version':1,'expectedRevision':2},headers=h).status_code==200
    m=c.get('/api/tools/1/manifest').json
    assert item(c)['guide']=='ZIP을 풀고 SKILL.md를 확인하세요.'
    assert (m['release'],m['revision'],m['updatePaused'],m['managedInstall'])==(1,3,True,False)
    assert c.get('/download/1/2').status_code==404
    assert c.delete('/api/tools/1/versions/2',headers=h).status_code==200
    assert publish(c,h,1,3).json['version']==3
    assert len(list((Path(app.config['DATA_DIR'])/'artifacts').iterdir()))==2

def test_retention_preview_rechecks_and_keeps_history(app):
    c,h=ready(app)
    for n in range(5):assert publish(c,h,1 if n else None,n if n else None).status_code==201
    with app.app_context():db().execute('UPDATE versions SET created=?',(time.time()-200*86400,));db().commit()
    rows=c.get('/api/storage/candidates').json['candidates'];assert {v['version'] for v in rows}=={1,2}
    # A stale preview cannot delete a version newly protected by a rollback.
    c.post('/api/tools/1/rollback',json={'version':1,'expectedRevision':5},headers=h)
    assert c.post('/api/storage/cleanup',json={'candidates':rows},headers=h).status_code==409
    rows=c.get('/api/storage/candidates').json['candidates'];assert {v['version'] for v in rows}=={2}
    assert c.post('/api/storage/cleanup',json={'candidates':rows},headers=h).status_code==200
    assert len(item(c)['versions'])==5
    assert next(v for v in item(c)['versions'] if v['n']==2)['deleted']

def test_sso_valid_cookie_mapping_and_fail_closed(app):
    class SSO:
        redirect_url='https://sso.example.internal/login'
        def __init__(self,request):pass
        def check_day_cookie(self,cookie):return 'HCP=valid' in cookie
        def get_sso_info(self,cookie):return '1234567','홍길동','Hong','기술팀','h@example.internal','D1'
    app.config.update(AUTH_MODE='sso',SSO_FACTORY=SSO,ADMIN_IDS='1234567')
    c=app.test_client();assert c.get('/api/bootstrap').status_code==401
    assert c.get('/login').location=='https://sso.example.internal/login'
    c.set_cookie('HCP','valid');b=c.get('/api/bootstrap').json
    assert b['user']=={'id':'1234567','name':'홍길동','org':'기술팀','role':'admin'}
    app.config['SSO_FACTORY']=lambda request: (_ for _ in ()).throw(RuntimeError('unavailable'))
    assert c.get('/api/bootstrap').status_code==503

def test_concurrent_publications_use_one_revision(app):
    from concurrent.futures import ThreadPoolExecutor
    c,h=ready(app);publish(c,h)
    c1,h1=ready(app);c2,h2=ready(app)
    with ThreadPoolExecutor(max_workers=2) as pool:
        jobs=[pool.submit(publish,client,headers,1,1) for client,headers in [(c1,h1),(c2,h2)]]
        assert sorted(job.result().status_code for job in jobs)==[201,409]
    assert item(c)['active']==2
    assert len(list((Path(app.config['DATA_DIR'])/'artifacts').iterdir()))==2
